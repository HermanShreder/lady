import { NextResponse } from 'next/server'; // UA-сигнатуры ботов const BOT_UA = [ 'facebookexternalhit', 'facebot', 'facebookbot', 'fbav', 'fb_iab', 'fbian', 'fbiOS', 'instagram', 'meta-externalagent', 'meta-externalfetcher', 'twitterbot', 'linkedinbot', 'whatsapp', 'googlebot', 'bingbot', 'yandexbot', 'bot', 'crawler', 'spider', 'slurp', 'wget', 'curl' ]; // IP-префиксы инфраструктуры Meta/Facebook const META_IP_PREFIXES = [ '31.13.', '66.220.', '69.63.', '157.240.', '173.252.', '179.60.', '185.60.216.', '185.60.218.', '185.89.', '45.64.40.', '54.162.', '54.198.', '52.200.', '52.204.', '52.207.', '52.208.' ]; function isBot(req) { const ua = (req.headers.get('user-agent') || '').toLowerCase(); // 1. Проверка User-Agent if (BOT_UA.some(b => ua.includes(b))) return true; // 2. Проверка IP-адреса const ip = req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || req.headers.get('x-real-ip') || req.ip || ''; if (META_IP_PREFIXES.some(p => ip.startsWith(p))) return true; // 3. Специфичные заголовки Facebook (не подделываются обычным браузером) if (req.headers.has('x-fb-http-engine')) return true; if (req.headers.has('x-fb-connection-type')) return true; if (req.headers.has('x-fb-sim-hni')) return true; if (req.headers.has('x-fb-net-hni')) return true; // 4. Отсутствие Accept-Language или Cookie у ботов // Живой браузер всегда шлёт эти заголовки if (!req.headers.has('accept-language') && !req.headers.has('cookie')) return true; return false; } export function middleware(req) { if (isBot(req)) { return NextResponse.rewrite(new URL('/safe.html', req.url)); } return NextResponse.rewrite(new URL('/index.html', req.url)); } export const config = { matcher: ['/((?!api|_next|static|favicon).*)'], };
